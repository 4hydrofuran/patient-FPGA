cmake_minimum_required(VERSION 3.16)
project(qwen35_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT LLAMA_SOURCE OR NOT LLAMA_BUILD)
  message(FATAL_ERROR "Provide fixed LLAMA_SOURCE and LLAMA_BUILD")
endif()
add_executable(qwen35-cpu main.cpp)
target_include_directories(qwen35-cpu PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include" "${LLAMA_SOURCE}/vendor/nlohmann")
target_link_libraries(qwen35-cpu PRIVATE "${LLAMA_BUILD}/bin/libllama.so")
target_compile_options(qwen35-cpu PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra)
set_target_properties(qwen35-cpu PROPERTIES BUILD_RPATH "${LLAMA_BUILD}/bin")
