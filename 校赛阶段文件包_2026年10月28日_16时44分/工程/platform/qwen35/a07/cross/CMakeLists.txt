project(a07_a53_cpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  message(FATAL_ERROR "This entry requires the matched AArch64 SDK")
endif()
set(ROOT /mnt/c/Patientqwen CACHE PATH "Project root")
set(LLAMA_SOURCE /home/member-a/qwen35-a05/source-v3 CACHE PATH "A05 preserved source")
if(NOT LLAMA_BUILD)
  message(FATAL_ERROR "Pass the separate A53 llama build")

find_library(CRYPTO_LIBRARY crypto REQUIRED)
add_library(sp_linear_cpu SHARED ${ROOT}/runtime/qwen35_w4a8/cpu_backend.cpp ${ROOT}/quant/reference/w4a8.cpp)
target_include_directories(sp_linear_cpu PRIVATE ${ROOT}/contracts/qwen35 ${ROOT}/quant/reference ${LLAMA_SOURCE}/vendor/nlohmann)
